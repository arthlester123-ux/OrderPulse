Database design

SQLite · table orders (one row per order line) · view v_net_sales · 5 indexes · table upload_log

CREATE TABLE orders (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    order_id TEXT NOT NULL UNIQUE,
    customer TEXT NOT NULL,  product TEXT NOT NULL,  category TEXT NOT NULL,
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    price    REAL    NOT NULL CHECK (price >= 0),
    order_date TEXT NOT NULL,                              -- ISO date
    status TEXT NOT NULL CHECK (status IN ('Pending','Processing','Shipped','Delivered','Cancelled')),
    line_total REAL NOT NULL                               -- quantity * price
);
CREATE INDEX idx_orders_date ON orders(order_date);   -- also: status, category, product, customer
CREATE VIEW v_net_sales AS SELECT * FROM orders WHERE status <> 'Cancelled';
